// ==UserScript==
// @name         TWdbUP — Inventario
// @namespace    https://github.com/TWdbUP/Database-TheWest
// @version      0.3.1
// @description  Pubblica il tuo inventario The West dalla finestra Script con un clic. Link valido 30 giorni.
// @include      https://*.the-west.*/game.php*
// @run-at       document-idle
// @noframes
// @grant        GM_xmlhttpRequest
// @grant        GM.xmlHttpRequest
// @grant        unsafeWindow
// @connect      import-inventory-tw.twdbup.workers.dev
// ==/UserScript==

/*
 * TWdbUP Inventario - versione pubblica con integrazione nella finestra Script del gioco.
 * La pubblicazione e' esplicita: solo clic su "Pubblica online".
 * Inventario pubblico a chi possiede il link, conservato per 30 giorni.
 * Gli accessi al sito richiedono una connessione Internet; le immagini vengono dal CDN.
 * Ricorda: ogni pubblicazione genera un NUOVO link, non aggiorna il vecchio.
 */
(() => {
  'use strict';
  const BASE = 'https://import-inventory-tw.twdbup.workers.dev';
  const ROOT_ID='twdbup-inventory-script-v03';
  // Secondo controllo: esegui SOLO sui mondi di gioco, nella vera pagina game.php.
  // Non attivarti su www.the-west.it, wiki.the-west.it o altre sottopagine.
  const worldHost = /^[a-z]{2,6}\d{1,4}\.the-west\.(?:[a-z]{2,3}|com\.(?:br|pt))$/i.test(location.hostname);
  const gamePath  = /^\/game\.php\/?$/i.test(location.pathname);
  if (!worldHost || !gamePath) return;
  if (window.top !== window.self) return;
  const getPage = () => {
    try { if(typeof unsafeWindow !== 'undefined' && unsafeWindow?.Bag) return unsafeWindow; } catch(_) {}
    try { if(window.wrappedJSObject?.Bag) return window.wrappedJSObject; } catch(_) {}
    return window;
  };

  function collectInventory() {
    const game = getPage();
    const Bag = game.Bag;
    const Wear = game.Wear;
    const ItemManager = game.ItemManager;
    if (!Bag || !Wear || !ItemManager || !Bag.loaded || !Bag.items_by_id) {
      throw new Error('Inventario non pronto: attendi il caricamento della Bag nel gioco e riprova.');
    }
    if (typeof Wear.get !== 'function' || typeof ItemManager.get !== 'function') {
      throw new Error('API Bag/Wear/ItemManager non disponibili: verifica il gestore userscript.');
    }

    // Le righe sottostanti provengono dal TEST 06B già collaudato.
    const SLOTS = ['head', 'neck', 'body', 'belt', 'pants', 'foot',
                 'left_arm', 'right_arm', 'yield', 'animal'];
  const LABELS = {
    head: 'Copricapi', neck: 'Collane', body: 'Giacche', belt: 'Cinture',
    pants: 'Pantaloni', foot: 'Scarpe', left_arm: 'Armi sinistre',
    right_arm: 'Armi destre', yield: 'Prodotti', animal: 'Animali',
    recipe: 'Ricette'
  };
  const ATTRS = {strength: 'Forza', flexibility: 'Agilità',
                 dexterity: 'Destrezza', charisma: 'Carisma'};
  const SKILLS = {
    build: 'Costruzione', punch: 'Vigore', tough: 'Robustezza',
    endurance: 'Energia', health: 'Punti vita', ride: 'Cavalcata',
    reflex: 'Riflessi', dodge: 'Schivata', hide: 'Nascosto', swim: 'Nuoto',
    aim: 'Mira', shot: 'Sparare', pitfall: 'Trappole',
    finger_dexterity: 'Precisione', repair: 'Riparazione',
    leadership: 'Comando', tactic: 'Tattica', trade: 'Negoziazione',
    animal: 'Rapporto con gli animali', appearance: 'Aspetto'
  };
  const getNum = x => Number(x) || 0;
  const fmt = n => {
    const v = Number(n);
    if (!Number.isFinite(v)) return String(n);
    // Sufficienti decimali per preservare i bonus originali degli oggetti.
    return String(Number(v.toFixed(9))).replace('.', ',');
  };

  // Bonus diretti (bonus.attributes / bonus.skills): gia' migliorati dal gioco.
  // Bonus dell'array bonus.item: l'ItemManager conserva i coefficienti BASE.
  // Applichiamo qui SOLO a essi la funzione nativa del gioco, senza usare
  // il livello del personaggio e senza alterare la definizione originale.
  function readBonuses(obj) {
    const b = obj.bonus || {};
    const up = Number(obj.getItemLevel?.() ?? obj.item_level ?? 0);
    const rows = [];
    for (const [key, value] of Object.entries(b.attributes || {})) {
      const n = Number(value);
      if (!Number.isFinite(n) || !n) continue;
      rows.push({kind: 'attr', text: (n < 0 ? '- ' : '+ ') + fmt(Math.abs(n)) + ' ' + (ATTRS[key] || key)});
    }
    for (const [key, value] of Object.entries(b.skills || {})) {
      const n = Number(value);
      if (!Number.isFinite(n) || !n) continue;
      rows.push({kind: 'skill', text: (n < 0 ? '- ' : '+ ') + fmt(Math.abs(n)) + ' ' + (SKILLS[key] || key)});
    }
    for (const entry of Array.isArray(b.item) ? b.item : []) {
      if (!entry) continue;
      const inner = entry.bonus || {};
      const rawField = inner.value != null ? inner.value : entry.value;
      const raw = rawField == null ? NaN : Number(rawField);
      const desc = String(entry.desc || inner.desc || '').trim();
      const kind = inner.type === 'attribute' ? 'attr'
                 : inner.type === 'skill' ? 'skill' : 'special';
      const fallbackLabel = (inner.type === 'attribute' ? ATTRS[inner.name] :
                            inner.type === 'skill' ? SKILLS[inner.name] : null) ||
                            inner.name || entry.type || '';
      let text;
      if (Number.isFinite(raw)) {
        let adjusted = raw;
        if (up > 0) {
          if (typeof ItemManager.calculateItemLevelBonus !== 'function') {
            throw Error('Funzione nativa bonus UP non disponibile: esportazione annullata.');
          }
          // Verificato sul cappello di John: 0,04 -> 0,048 e 300 -> 360 a UP2.
          const delta = Number(ItemManager.calculateItemLevelBonus(up, raw));
          if (!Number.isFinite(delta)) throw Error('Miglioramento UP non valido per oggetto ' + obj.item_id);
          adjusted = raw + delta;
        }
        // Conserva la descrizione originale e sostituisce soltanto il numero.
        // Gestisce anche descrizioni del tipo '+-1.96 Agilita''.
        const m = desc.match(/^\s*[+-]\s*(-?[\d.,]+)\s*(%?)(.*)$/s);
        if (m) {
          const isPercent = m[2] === '%';
          const shownBase = Number(m[1].replace(',', '.'));
          // I dati percentuali di The West sono normalmente in frazioni
          // (0.1 == 10%). Non applicare il *100 a valori gia' in percentuale.
          const fractional = isPercent && Number.isFinite(shownBase) &&
            Math.abs(Math.abs(shownBase) - Math.abs(raw * 100)) < 1e-6;
          const amount = isPercent && fractional ? adjusted * 100 : adjusted;
          text = (amount < 0 ? '- ' : '+ ') + fmt(Math.abs(amount)) +
                 (isPercent ? '%' : '') + ' ' + m[3].trim();
        } else {
          // Nessun numero iniziale (es. 'Ripresa immediata dopo KO').
          text = (adjusted < 0 ? '- ' : '+ ') + fmt(Math.abs(adjusted)) + ' ' + (desc || fallbackLabel);
        }
      } else {
        text = desc || fallbackLabel;
      }
      if (text) rows.push({kind, text});
    }
    return rows;
  }

  const items = new Map();
  let missing = 0;
  function add(itemId, qty, source, objHint) {
    const id = Number(itemId);
    if (!Number.isSafeInteger(id) || qty <= 0) return;
    const obj = objHint || ItemManager.get(id);
    if (!obj) { missing++; return; }
    let row = items.get(id);
    if (!row) {
      row = {
        id, baseId: Number(obj.item_base_id) || 0,
        name: String(obj.name || ('Oggetto ' + id)),
        type: String(obj.type || 'altro'),
        up: Number(obj.item_level) || 0,
        bag: 0, wear: 0,
        price: getNum(obj.price), sellPrice: getNum(obj.sell_price),
        auctionable: obj.auctionable === true,
        tradeable: obj.tradeable === true,
        sellable: obj.sellable === true,
        upgradeable: obj.upgradeable === true,
        level: getNum(obj.level),
        image: String(obj.image || ''),
        bonuses: readBonuses(obj)
      };
      items.set(id, row);
    }
    row[source] += qty;
  }

  // Una sola volta per item_id: non percorrere items_by_inv_id,
  // poiché alcune chiavi inv_id puntano allo stesso stack!
  for (const [key, inv] of Object.entries(Bag.items_by_id || {})) {
    if (!inv) continue;
    const id = Number(key);
    add(id, getNum(inv.count), 'bag', inv.obj || ItemManager.get(id));
  }
  // Ogni slot indossato è un pezzo fisico aggiuntivo.
  for (const slot of SLOTS) {
    const worn = Wear.get(slot);
    if (!worn) continue;
    const id = Number(worn.getId());
    add(id, 1, 'wear', ItemManager.get(id));
  }

  // Ordinamento di visualizzazione fisso per ID, senza pulsanti ordina.
  // Gli oggetti indossati non restano in fondo alle rispettive categorie.
  const rows = Array.from(items.values()).map(r => ({...r, count: r.bag + r.wear}));
  rows.sort((a, b) => a.id - b.id);
  const total = field => rows.reduce((acc, r) => acc + getNum(r[field]), 0);
  const summary = {
    bag: total('bag'), wear: total('wear'),
    pieces: total('count'), unique: rows.length,
    buy: rows.reduce((s, r) => s + r.count * r.price, 0),
    sell: rows.reduce((s, r) => s + r.count * r.sellPrice, 0)
  };
  const time = new Date();
  const world = location.hostname || 'mondo';
  const payload = {v: 2, date: time.toISOString(), world, summary, rows};


    if (missing > 0) throw new Error('Definizioni mancanti: '+missing+'. Pubblicazione annullata per evitare inventari incompleti.');
    return {payload,summary,missing};
  }

  const hasClassic = typeof GM_xmlhttpRequest === 'function';
  const hasModern = typeof GM !== 'undefined' && typeof GM.xmlHttpRequest === 'function';
  function requestGM(method,url,data,headers={}) {
    if (!hasClassic && !hasModern) return Promise.reject(Error('Gestore userscript senza GM_xmlhttpRequest'));
    return new Promise((resolve,reject)=>{
      let finished=false;
      const yes=v=>{if(!finished){finished=true;resolve(v);}};
      const no=v=>{if(!finished){finished=true;reject(v instanceof Error?v:Error(String(v)));}};
      try {
        const opts={method,url,headers,data,anonymous:true,timeout:30000,responseType:'text',
          onload:yes,onerror:()=>no(Error('Connessione Worker non riuscita')),
          ontimeout:()=>no(Error('Timeout di collegamento')),onabort:()=>no(Error('Richiesta annullata'))};
        const ret=hasClassic?GM_xmlhttpRequest(opts):GM.xmlHttpRequest(opts);
        if(ret&&typeof ret.then==='function')ret.then(yes,no);
      }catch(e){no(e);}
    });
  }
  function parseResponse(r) {
    let data;try{data=JSON.parse(r.responseText||r.response||'{}');}
    catch(_){throw Error('Risposta del servizio non in formato JSON (HTTP '+r.status+')');}
    if(r.status!==200||!data.ok)throw Error(data.error||'Errore HTTP '+r.status);
    return data;
  }
  async function gzipJSON(payload) {
    if (typeof CompressionStream!=='function') throw Error('Browser senza CompressionStream');
    const bytes=new TextEncoder().encode(JSON.stringify(payload));
    if(bytes.byteLength>6*1024*1024)throw Error('Inventario superiore al limite JSON di 6 MiB');
    const compressed=await new Response(new Blob([bytes]).stream().pipeThrough(new CompressionStream('gzip'))).arrayBuffer();
    if(compressed.byteLength>750*1024)throw Error('Inventario compresso superiore al limite di 750 KiB');
    return {compressed,rawBytes:bytes.byteLength};
  }
  function saveJSON(payload) {
    const json=JSON.stringify(payload);
    const url=URL.createObjectURL(new Blob([json],{type:'application/json'}));
    const a=document.createElement('a');
    a.href=url;a.download='TWdbUP_Inventario_'+payload.world+'_'+payload.date.replace(/[:.]/g,'-')+'.json';
    a.style.display='none';document.body.appendChild(a);a.click();a.remove();
    setTimeout(()=>URL.revokeObjectURL(url),5000);
    return a.download;
  }
  // TEST 14: The West dispone della propria finestra Script.
  // TheWestApi.register(...).setGui(...) crea una scheda senza pannelli fissi.
  // Il contenuto HTML viene ricreato dal gioco; i clic sono gestiti con delega
  // sul documento, quindi continuano a funzionare anche riaprendo la finestra.
  const TAB_NAME='TWdbUP Inventario';
  const HTML = `<div id="${ROOT_ID}" style="max-width:570px;margin:18px auto;padding:14px 18px;background:#e8d4ac;border:1px solid #8a6946;border-radius:6px;color:#3c2615;font:14px Georgia,serif;text-align:center;box-sizing:border-box;">
    <h2 style="margin:3px 0 10px;font-size:23px;color:#573015;">TWdbUP · Inventario</h2>
    <p style="margin:5px 0 16px;font-size:13px;">Condividi l'inventario del tuo personaggio con un clic.</p>
    <div style="display:flex;gap:10px;justify-content:center;flex-wrap:wrap;">
      <button type="button" data-twdbup-action="publish" style="cursor:pointer;padding:10px 15px;background:#ead085;color:#32200c;border:1px solid #6b4625;border-radius:4px;font-weight:bold;">Pubblica online · 1 clic</button>
      <button type="button" data-twdbup-action="backup" style="cursor:pointer;padding:10px 15px;background:#f1ddb2;color:#32200c;border:1px solid #6b4625;border-radius:4px;font-weight:bold;">Esporta JSON (riserva)</button>
    </div>
    <div data-twdbup-status style="margin-top:12px;font:12px Arial,sans-serif;overflow-wrap:anywhere;">Link condivisibile per 30 giorni.</div>
    <div data-twdbup-result hidden style="margin:12px auto 0;padding:9px;border:1px solid #8f724e;background:#f4e3bf;max-width:340px;">
      <a data-twdbup-link href="#" target="_blank" rel="noopener noreferrer" style="font-weight:bold;">Apri inventario</a>
      <button type="button" data-twdbup-action="copy" style="display:block;margin:8px auto 0;cursor:pointer;background:#e9c776;color:#352316;border:1px solid #6b4625;border-radius:4px;padding:5px 12px;">Copia link</button>
    </div>
    <p style="margin:14px 0 2px;font:11px Arial,sans-serif;color:#725536;">L'inventario è pubblico per chi conosce il link e scade dopo 30 giorni.</p>
  </div>`;

  let busy=false;
  let publishedLink='';
  const all = selector => Array.from(document.querySelectorAll('#'+ROOT_ID+' '+selector));
  const setStatus = message => all('[data-twdbup-status]').forEach(el=>el.textContent=message);
  const setBusy = value => {
    busy=value;
    all('[data-twdbup-action="publish"],[data-twdbup-action="backup"]').forEach(el=>el.disabled=value);
  };
  const setResult = url => {
    publishedLink=url;
    all('[data-twdbup-result]').forEach(el=>el.hidden=!url);
    all('[data-twdbup-link]').forEach(el=>{el.href=url||'#';});
  };
  const copyLink = () => {
    if(!publishedLink)return;
    if(navigator.clipboard?.writeText){
      navigator.clipboard.writeText(publishedLink)
        .then(()=>setStatus('Link copiato'))
        .catch(()=>prompt('Copia il link',publishedLink));
    }else prompt('Copia il link',publishedLink);
  };
  async function publishFromGame() {
    // Apertura subito nel gesto dell'utente: evita i popup blocker.
    const tab=window.open('about:blank','_blank');
    if(tab){try{tab.opener=null;}catch(_) {}}
    setBusy(true);setResult('');setStatus('Preparazione inventario...');
    try{
      const {payload,summary}=collectInventory();
      setStatus('Comprimo '+summary.unique.toLocaleString('it-IT')+' oggetti...');
      const {compressed,rawBytes}=await gzipJSON(payload);
      setStatus('Invio su Cloudflare...');
      const health=parseResponse(await requestGM('GET',BASE+'/health?test14='+Date.now()));
      if(health.directUploadTest!==true)throw Error('Servizio di pubblicazione diretta non disponibile');
      const response=await requestGM('POST',BASE+'/api/direct/upload',compressed,{
        'Content-Type':'application/octet-stream',
        'X-TWUP-Format':'gzip-v1',
        'X-TWDBUP-Client':'userscript-v02-test'
      });
      const data=parseResponse(response);
      const target=new URL(data.link,BASE);
      if(target.origin!==BASE||!/^\/i\/[A-Za-z0-9_-]{24}$/.test(target.pathname))throw Error('Link restituito non valido');
      setResult(target.href);
      setStatus('Pubblicato! '+summary.pieces.toLocaleString('it-IT')+' pezzi · Scade tra 30 giorni');
      console.log('[TWdbUP] Pubblicazione TEST 14 OK',{world:payload.world,bag:summary.bag,wear:summary.wear,pieces:summary.pieces,unique:summary.unique,jsonBytes:rawBytes,gzipBytes:compressed.byteLength,link:target.href,expiresAt:data.expiresAt});
      if(tab){tab.location.href=target.href;}
    }catch(e){
      if(tab){try{tab.close();}catch(_) {}}
      setStatus('ERRORE: '+(e.message||e));
      console.error('[TWdbUP] Pubblicazione TEST 14 fallita:',e);
    }finally{setBusy(false);}
  }
  function backupFromGame() {
    setBusy(true);
    try{
      const {payload,summary}=collectInventory();
      const file=saveJSON(payload);
      setStatus('Scaricato JSON: '+summary.unique.toLocaleString('it-IT')+' oggetti');
      console.log('[TWdbUP] Backup locale',file,summary);
    }catch(e){
      console.error('[TWdbUP] Backup:',e);
      setStatus('ERRORE: '+(e.message||e));
    }finally{setBusy(false);}
  }
  document.addEventListener('click',event=>{
    const target=event.target;
    if(!(target instanceof Element))return;
    const button=target.closest('[data-twdbup-action]');
    if(!button || !button.closest('#'+ROOT_ID))return;
    event.preventDefault();
    event.stopPropagation();
    const action=button.getAttribute('data-twdbup-action');
    if(action==='copy'){copyLink();return;}
    if(busy)return;
    if(action==='backup')backupFromGame();
    if(action==='publish')void publishFromGame();
  },true);

  let registered=false;
  function registerTab() {
    if(registered)return true;
    const game=getPage();
    if(typeof game.TheWestApi?.register!=='function')return false;
    try{
      const entry=game.TheWestApi.register('TWDBUP_INVENTORY',TAB_NAME,'2.0','99','TWdbUP','https://twdbup.github.io/Database-TheWest/inventari/');
      if(!entry || typeof entry.setGui!=='function')throw Error('TheWestApi.setGui non disponibile');
      entry.setGui(HTML);
      registered=true;
      console.log('[TWdbUP] TEST 14: scheda registrata nella finestra Script. Nessun pannello fisso.');
      return true;
    }catch(e){
      // Evitiamo registrazioni ripetute se l'API ha già registrato il nostro ID.
      registered=true;
      console.error('[TWdbUP] TEST 14: integrazione finestra Script non riuscita:',e);
      return true;
    }
  }
  // Aspetta che il gioco abbia inizializzato TheWestApi, senza modificare il menu.
  if(!registerTab()){
    let tries=0;
    const timer=setInterval(()=>{
      tries++;
      if(registerTab()||tries>=180){
        clearInterval(timer);
        if(!registered)console.warn('[TWdbUP] TEST 14: TheWestApi non trovata dopo 90 secondi. Ricarica il mondo e verifica che la finestra Script sia disponibile.');
      }
    },500);
  }
})();
