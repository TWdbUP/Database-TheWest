  function readBonuses(obj) {
    const b = obj.bonus || {};
    const up = Number(obj.getItemLevel?.() ?? obj.item_level ?? 0);
    const rows = [];
    for (const [key, value] of Object.entries(b.attributes || {})) {

    }

        // Gestisce anche descrizioni del tipo '+-1.96 Agilita''.

        }
      } else {
        text = desc || fallbackLabel;
      }

    }
    return rows;
  }

  function normalizedKey(raw){
    if(validKey(raw))return raw;
    if(typeof raw!=='string')return null;
    const trimmed=raw.trim();
    if(validKey(trimmed))return trimmed;
    try{
      const parsed=JSON.parse(trimmed);
      if(validKey(parsed))return parsed;
    }catch(_){}
    return null;
  }
  function describeSaved(raw){
    if(raw===''||raw===null||raw===undefined)return 'nessuna chiave';

    const size=typeof raw==='string'?' · lunghezza '+raw.length:'';
    return 'tipo '+kind+size;

  // Codifica Base64url manuale, indipendente da btoa() dell'ambiente di gioco.
  // E' importante usare SEMPRE getRandomValues crittografico: niente Math.random.
  function base64url32(raw){
    const abc='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_';
    let out='';
    for(let i=0;i<raw.length;i+=3){
      const a=raw[i], b=(i+1<raw.length?raw[i+1]:0), c=(i+2<raw.length?raw[i+2]:0);
      out+=abc[a>>>2];
      out+=abc[((a&3)<<4)|(b>>>4)];
      if(i+1<raw.length)out+=abc[((b&15)<<2)|(c>>>6)];
      if(i+2<raw.length)out+=abc[c&63];
    }
    return out;
  }
  function createKey(){
    const random=(typeof crypto!=='undefined'&&typeof crypto.getRandomValues==='function')?crypto:null;
    if(!random)throw Error('Web Crypto non disponibile nel gestore userscript: nessuna chiave salvata');
    const raw=new Uint8Array(32);
    random.getRandomValues(raw);
    if(raw.length!==32)throw Error('Generatore casuale non valido: lunghezza inattesa, nessuna chiave salvata');
    const key=base64url32(raw);
    if(!validKey(key))throw Error('Codifica chiave non valida (lunghezza '+key.length+'): nessuna modifica alla memoria');
    return key;
  }
  async function getPrivateKey(payload){
    const name=keyName(payload);
    const saved=await storedGet(name);
    if(saved!==''&&saved!==null&&saved!==undefined){
      const usable=normalizedKey(saved);
      if(!usable)throw Error('Accesso locale non valido ('+describeSaved(saved)+'). Nessun dato cancellato: apri Trasferisci accesso e ripristina il file di backup.');
      if(saved!==usable){
        await storedSet(name,usable);
        if(await storedGet(name)!==usable)throw Error('Normalizzazione chiave non persistente. Nessun nuovo link creato.');
      }
      return usable;
    }
    const key=createKey();
    await storedSet(name,key);
    if(await storedGet(name)!==key)throw Error('Chiave privata non persistente: pubblicazione annullata');
    return key;
  }

        const ret=hasClassic?GM_xmlhttpRequest(opts):GM.xmlHttpRequest(opts);
        if(ret&&typeof ret.then==='function')ret.then(yes,no);

  async function gzipJSON(payload) {
    if (typeof CompressionStream!=='function') throw Error('Browser senza CompressionStream');
    const bytes=new TextEncoder().encode(JSON.stringify(payload));
    if(bytes.byteLength>6*1024*1024)throw Error('Inventario superiore al limite JSON di 6 MiB');
    const compressed=await new Response(new Blob([bytes]).stream().pipeThrough(new CompressionStream('gzip'))).arrayBuffer();
    if(compressed.byteLength>750*1024)throw Error('Inventario compresso superiore al limite di 750 KiB');
    return {compressed,rawBytes:bytes.byteLength};
  }

      setStatus('Inventario pubblicato/aggiornato! '+summary.pieces.toLocaleString('it-IT')+' pezzi · Scade tra 30 giorni');
      console.log('[TWdbUP] Pubblicazione OK',{world:payload.world,bag:summary.bag,wear:summary.wear,pieces:summary.pieces,unique:summary.unique,jsonBytes:rawBytes,gzipBytes:compressed.byteLength,link:target.href,expiresAt:data.expiresAt});
      if(tab){tab.location.href=target.href;}
    }catch(e){
      if(tab){try{tab.close();}catch(_) {}}
      setStatus('ERRORE: '+(e.message||e));
      console.error('[TWdbUP] Pubblicazione fallita:',e);
    }finally{setBusy(false);}
  }
  function backupFromGame() {
    setBusy(true);
    try{
      const {payload,summary}=collectInventory();
      const file=saveJSON(payload);
      setStatus('Scaricato JSON: '+summary.unique.toLocaleString('it-IT')+' oggetti');
      console.log('[TWdbUP] Backup locale',file,summary);
    }catch(e){
      console.error('[TWdbUP] Backup:',e);
      setStatus('ERRORE: '+(e.message||e));
    }finally{setBusy(false);}
  }
  async function handleAccessAction(action,file){
    setBusy(true);
    try{
      if(action==='accessbackup')await backupPrivateKey();
      else if(action==='accessrestore')await restorePrivateKeyFile(file);
    }catch(e){setStatus('ERRORE: '+(e.message||e));console.error('[TWdbUP] Trasferimento accesso:',e);}
    finally{setBusy(false);}
  }
  document.addEventListener('click',event=>{
    const target=event.target;
    if(!(target instanceof Element))return;
    const button=target.closest('[data-twdbup-action]');
    if(!button || !button.closest('#'+ROOT_ID))return;
    event.preventDefault();
    event.stopPropagation();
    const action=button.getAttribute('data-twdbup-action');
    if(action==='copy'){copyLink();return;}
    if(busy)return;
    if(action==='backup')backupFromGame();
    if(action==='publish')void publishFromGame();
    if(action==='accessbackup')void handleAccessAction(action);
    if(action==='accessrestore'){const input=all('[data-twdbup-restore-file]')[0];if(input){input.value='';input.click();}}
  },true);

  document.addEventListener('change',event=>{
    const input=event.target;
    if(!(input instanceof HTMLInputElement)||!input.matches('#'+ROOT_ID+' [data-twdbup-restore-file]'))return;
    if(busy||!input.files?.[0])return;
    void handleAccessAction('accessrestore',input.files[0]);
  },true);

  let registered=false;
  function registerTab() {
    if(registered)return true;
    const game=getPage();
    if(typeof game.TheWestApi?.register!=='function')return false;
    try{
      const entry=game.TheWestApi.register('TWDBUP_INVENTORY',TAB_NAME,'2.0','99','TWdbUP','https://twdbup.github.io/Database-TheWest/inventari/');
      if(!entry || typeof entry.setGui!=='function')throw Error('TheWestApi.setGui non disponibile');
      entry.setGui(HTML);
      registered=true;
      console.log('[TWdbUP] Inventario integrato nella finestra Script.');
      return true;
    }catch(e){
      // Evitiamo registrazioni ripetute se l'API ha già registrato il nostro ID.
      registered=true;
      console.error('[TWdbUP] Integrazione finestra Script non riuscita:',e);
      return true;
    }
  }
  // Aspetta che il gioco abbia inizializzato TheWestApi, senza modificare il menu.
  if(!registerTab()){
    let tries=0;
    const timer=setInterval(()=>{
      tries++;
      if(registerTab()||tries>=180){
        clearInterval(timer);
        if(!registered)console.warn('[TWdbUP] TheWestApi non trovata dopo 90 secondi. Ricarica la pagina del gioco.');
      }
    },500);
  }
})();
